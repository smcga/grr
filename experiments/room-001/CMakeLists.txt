cmake_minimum_required(VERSION 3.22)
project(room001 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(room001 room001.cpp)
if(MSVC)
  target_compile_options(room001 PRIVATE /O2 /W4)
else()
  target_compile_options(room001 PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME physical_checks COMMAND room001 --test)
