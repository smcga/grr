cmake_minimum_required(VERSION 3.22)
project(Room001Audition VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(JUCE_PATH)
  add_subdirectory("${JUCE_PATH}" JUCE)
else()
  find_package(JUCE CONFIG QUIET)
  if(NOT JUCE_FOUND)
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
  endif()
endif()
juce_add_plugin(Room001Audition COMPANY_NAME "Ghost Room Rewrites" BUNDLE_ID com.ghostroomrewrites.room001 PLUGIN_MANUFACTURER_CODE GhRw PLUGIN_CODE Rm01 FORMATS VST3 Standalone PRODUCT_NAME "GRR ROOM-001" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(Room001Audition PRIVATE Plugin.cpp)
target_compile_definitions(Room001Audition PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Room001Audition PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
juce_add_console_app(Room001Smoke PRODUCT_NAME "ROOM-001 smoke")
target_sources(Room001Smoke PRIVATE Smoke.cpp)
target_compile_definitions(Room001Smoke PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(Room001Smoke PRIVATE Room001Audition juce::juce_audio_utils)
enable_testing()
add_test(NAME processor_smoke COMMAND Room001Smoke)
